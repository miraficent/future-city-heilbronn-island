"use client";
import Link from "next/link";
import { useEffect,useRef,useState } from "react";
import type { Angebot } from "@/lib/angebot";
import type { Meta } from "@/lib/daten";
import { AUFGABEN_TITEL,neuerSpielstand,pruefeSpielstand,STUFEN,stufeFuerAlter,WUENSCHE,type Spielstand } from "@/lib/abenteuer";
import { alterAvatar,pruefeAvatar,STANDARD_AVATAR,type AvatarKonfiguration } from "@/lib/avatar";
import { LERNORTE } from "@/lib/lernwelt";
import { passendeSpielAngebote,STATIONEN } from "@/lib/spiel";
import { lies,schreibe,loesche } from "@/lib/speicher";
import { useSammlung } from "../sammlung";
import { Angebotskarte } from "../angebotskarte";
import { Avatar,AvatarKontext } from "./avatar";
import { AvatarBaukasten } from "./avatar-baukasten";
import { Stadtkarte } from "./stadtkarte";
import { AufgabenEinstieg } from "./aufgaben-einstieg";
import { Lernaufgabe } from "./lernaufgabe";
import { Rakete } from "./rakete";
import { Sprachknopf } from "./sprachknopf";

type Ansicht="elternhinweis"|"figur"|"anleitung"|"stadt"|"ort"|"aufgabe"|"finale";
type SitzungsAuswahl={avatar:AvatarKonfiguration;begonnen:boolean;elternhinweis:boolean;stumm:boolean;kursalterBestaetigt:boolean};
export function Lernreise({angebote,meta,jetzt:anfang}:{angebote:Angebot[];meta:Meta;jetzt:string}){
 const [geladen,setGeladen]=useState(false),[ansicht,setAnsicht]=useState<Ansicht>("elternhinweis"),[avatar,setAvatar]=useState<AvatarKonfiguration>(STANDARD_AVATAR),[begonnen,setBegonnen]=useState(false),[stumm,setStumm]=useState(false);
 const [kursalterBestaetigt,setKursalterBestaetigt]=useState(false);
 const [stand,setStand]=useState<Spielstand>(neuerSpielstand),[ortId,setOrtId]=useState("experimenta"),[phase,setPhase]=useState(0),[eingefuehrt,setEingefuehrt]=useState(false),[erfolg,setErfolg]=useState(false),[jetzt,setJetzt]=useState(anfang),[vollbild,setVollbild]=useState(false);
 const [fenster,setFenster]=useState<"tor"|"erwachsene"|"rucksack"|"hilfe"|undefined>(),[freigabe,setFreigabe]=useState(false),[halten,setHalten]=useState(false),[reset,setReset]=useState(false);
 const dialog=useRef<HTMLDialogElement>(null),haupt=useRef<HTMLElement>(null),timer=useRef<ReturnType<typeof setTimeout>|null>(null),fokus=useRef<HTMLElement|null>(null);
 const merk=useSammlung("kind"),stufe=stufeFuerAlter(stand.alter),erfolge=stand.fortschritt[stufe],ort=LERNORTE.find(o=>o.id===ortId)!,station=STATIONEN.find(s=>s.id===ortId)!;
 const teile=STATIONEN.filter(s=>[0,1,2].every(n=>erfolge.includes(`${s.id}-${n}`))).map(s=>s.teil),komplett=teile.length===4;
 const passende=kursalterBestaetigt?passendeSpielAngebote(angebote,station.thema,stand.alter,new Date(jetzt)):[];
 useEffect(()=>{
  const s=lies<Partial<SitzungsAuswahl>|null>("sitzung","lernreise.v1",null),alt=lies<unknown>("sitzung","spielwelt.figur",null);
  setKursalterBestaetigt(s?.kursalterBestaetigt===true);setAvatar(s?.avatar?pruefeAvatar(s.avatar):alterAvatar(alt));setStumm(s?.stumm===true);setBegonnen(s?.begonnen===true);
  setStand(pruefeSpielstand(lies("sitzung","abenteuer.v2",null)));
  if(s?.elternhinweis===true)setAnsicht(s.begonnen===true?"stadt":"figur");
  setJetzt(new Date().toISOString());setGeladen(true);
  return()=>{if(timer.current)clearTimeout(timer.current);};
 },[]);
 useEffect(()=>{if(geladen&&ansicht!=="elternhinweis")schreibe("sitzung","lernreise.v1",{avatar,begonnen,elternhinweis:true,stumm,kursalterBestaetigt});},[avatar,begonnen,stumm,kursalterBestaetigt,geladen,ansicht]);
 useEffect(()=>{document.dispatchEvent(new Event("cosmos:audio-stopp"));},[ansicht,phase,ortId]);
 useEffect(()=>{if(fenster){fokus.current=document.activeElement as HTMLElement;dialog.current?.showModal();}else{dialog.current?.close();fokus.current?.focus({preventScroll:true});}},[Boolean(fenster)]);
 useEffect(()=>{const voll=()=>setVollbild(Boolean(document.fullscreenElement));document.addEventListener("fullscreenchange",voll);return()=>document.removeEventListener("fullscreenchange",voll);},[]);
 function speichere(neu:Spielstand){setStand(neu);schreibe("sitzung","abenteuer.v2",neu);}
 function besuchen(id:string){setOrtId(id);setPhase([0,1,2].find(n=>!erfolge.includes(`${id}-${n}`))??0);setEingefuehrt(false);setErfolg(false);setAnsicht("ort");}
 function neuePhase(n:number){document.dispatchEvent(new Event("cosmos:audio-stopp"));setPhase(n);setErfolg(false);setEingefuehrt(false);}
 function geschafft(){const id=`${ortId}-${phase}`;if(!erfolge.includes(id))speichere({...stand,fortschritt:{...stand.fortschritt,[stufe]:[...erfolge,id]}});setErfolg(true);}
 function wunsch(id:string){speichere({...stand,wuensche:stand.wuensche.includes(id)?stand.wuensche.filter(x=>x!==id):[...stand.wuensche,id]});}
 function erwachsene(){setFenster(freigabe?"erwachsene":"tor");}
 function haltenStart(){if(timer.current)return;setHalten(true);timer.current=setTimeout(()=>{timer.current=null;setHalten(false);setFreigabe(true);setFenster("erwachsene");},3000);}
 function haltenEnde(){if(timer.current)clearTimeout(timer.current);timer.current=null;setHalten(false);}
 async function bildschirm(){if(vollbild){if(document.fullscreenElement)await document.exitFullscreen().catch(()=>{});setVollbild(false);}else{setVollbild(true);await haupt.current?.requestFullscreen?.().catch(()=>{});}}
 function neu(){speichere(neuerSpielstand());loesche("sitzung","lernreise.v1");loesche("sitzung","spielwelt.figur");loesche("sitzung","spielwelt.rakete");loesche("sitzung","spiel.erfolge");merk.leeren();setAvatar({...STANDARD_AVATAR});setBegonnen(false);setKursalterBestaetigt(false);setFreigabe(false);setReset(false);setFenster(undefined);setAnsicht("elternhinweis");}
 const karte=(a:Angebot)=><Angebotskarte key={a.id} angebot={a} meta={meta} modus="eltern" gemerkt={merk.ids.includes(a.id)} merken={merk.merken} jetzt={jetzt}/>;
 return <AvatarKontext.Provider value={avatar}><main ref={haupt} id="inhalt" className={`lernreise ${vollbild?"lernreise-vollbild":""}`} onKeyDown={e=>{if(e.key==="Escape"&&!document.fullscreenElement)setVollbild(false);}}>
 <header className="reise-kopf"><button className="reise-logo" onClick={()=>begonnen&&setAnsicht("stadt")} aria-label="COSMOS – zurück zur Heilbronn-Übersicht"><svg className="logo-sonne" aria-hidden="true" viewBox="0 0 40 40" width="40" height="40"><g fill="currentColor">{[0,60,120].map(r=><ellipse key={r} cx="20" cy="20" rx="6" ry="18" transform={`rotate(${r} 20 20)`}/>)}</g><circle cx="20" cy="20" r="5" fill="#f3efdf"/></svg><span>COSMOS<small>HEILBRONN ENTDECKEN</small></span></button><div className="reise-werkzeuge">{begonnen&&<button className="welt-chip avatar-wechsel" onClick={()=>setAnsicht("figur")}><Avatar mini/><span>Meine Figur</span></button>}<button className="welt-chip" onClick={()=>setFenster("hilfe")}>Hilfe</button><button className="welt-chip erwachsenen-knopf" onClick={erwachsene}>Für Erwachsene</button></div></header>
 {!geladen?<section className="reise-willkommen"><h1>Dein Heilbronn-Abenteuer.</h1><p>Die Stadt wird geöffnet.</p></section>:<>
 {ansicht==="elternhinweis"&&<section className="reise-willkommen"><div className="willkommen-bild"><img src="/medien/heilbronn/experimenta.png" alt="Gemalte experimenta in Heilbronn mit Dreiecksfassade und Wasser"/><Avatar wert={avatar}/><span className="ort-fahne">HEILBRONN</span></div><div className="willkommen-text"><p className="welt-augenbraue">DEINE STADT. DEINE IDEEN.</p><h1>Ein Abenteuer wartet vor deiner Tür.</h1><p>Vier echte Lernorte. Eine Rakete voller Ideen.</p><div className="gemeinsam-hinweis"><strong>Unter 16?</strong><span>Starte gemeinsam mit deinen Eltern.</span></div><button className="welt-knopf" onClick={()=>setAnsicht("figur")}>Wir starten gemeinsam</button><p className="reise-kleindruck">Ohne Konto. Ohne Namen. Dein Spiel bleibt in dieser Browsersitzung.</p></div></section>}
 {ansicht==="figur"&&<AvatarBaukasten wert={avatar} aendern={setAvatar} fertig={()=>setAnsicht(begonnen?"stadt":"anleitung")} zurueck={()=>setAnsicht(begonnen?"stadt":"elternhinweis")}/>}
 {ansicht==="anleitung"&&<section className="reise-anleitung"><p className="welt-augenbraue">DREI KLEINE SCHRITTE</p><h1>So beginnt deine Reise.</h1><div className="reise-bildanleitung"><div><img src="/medien/heilbronn/experimenta.png" alt="Ein Lernort"/><b>1</b><h2>Wähle einen Ort.</h2></div><div><svg viewBox="0 0 150 90" aria-hidden="true"><path d="M15 60h50V25h65" stroke="#548773" strokeWidth="9" fill="none"/><path d="M130 40V10l20 8-20 8" fill="#ca8753"/></svg><b>2</b><h2>Probiere eine Idee.</h2></div><div><svg viewBox="-60 -100 120 160" aria-hidden="true"><path d="M0-80Q-35-40-23 30h46Q35-40 0-80" fill="#f1e5c4" stroke="#416253" strokeWidth="3"/><circle cy="-15" r="16" fill="#97c0b9" stroke="#416253" strokeWidth="3"/><path d="m-23 5-21 35 24-7M23 5l21 35-24-7" fill="#cf7d5a"/><path d="m-10 32 10 25 10-25" fill="#e6b451"/></svg><b>3</b><h2>Baue unsere Rakete.</h2></div></div><div className="reise-stufen"><h2>Wie möchtest du starten?</h2>{STUFEN.map(s=><button className="welt-chip" key={s.id} aria-pressed={s.id===stufe} onClick={()=>{setKursalterBestaetigt(false);speichere({...stand,alter:s.min===6?8:s.min===9?10:14});}}>{s.name}<small>{s.alter}</small></button>)}</div><div className="welt-aktionen"><Sprachknopf id="willkommen" stumm={stumm}/><button className="welt-knopf" onClick={()=>{setBegonnen(true);setAnsicht("stadt");}}>Los geht’s, Heilbronn!</button></div></section>}
 {ansicht==="stadt"&&<section className="reise-stadt"><div className="reise-titel"><div><p className="welt-augenbraue">AM NECKAR GIBT ES VIEL ZU ENTDECKEN</p><h1>Hallo, Heilbronn.</h1><p>Welchen Ort besuchen wir heute?</p></div><div className="reise-fortschritt"><span>{erfolge.length}<small> / 12 Ideen</small></span><button className="welt-chip" onClick={()=>setFenster("rucksack")}>Mein Rucksack · {merk.ids.length}</button></div></div><div className="reise-stadt-layout"><Stadtkarte besuchen={besuchen} erfolge={erfolge}/><div className="reise-ortskarten">{LERNORTE.map((o,i)=><button className="reise-ortskarte" key={o.id} onClick={()=>besuchen(o.id)} style={{"--ort-farbe":o.farbe} as React.CSSProperties}><div className="ortskarte-bild"><img src={o.bild} alt=""/><span>{i+1}</span></div><div><small>{o.kurz}</small><h2>{o.name}</h2><span className="ortskarte-weiter">{erfolge.filter(id=>id.startsWith(o.id+"-")).length} / 3 Ideen <b aria-hidden="true">→</b></span></div></button>)}</div></div><section className="reise-rakete"><svg viewBox="-80 -130 160 190" aria-label={`${teile.length} von vier Raketenteilen`}><Rakete teile={teile} zustand="bereit"/></svg><div><h2>{komplett?"Alle Ideen sind an Bord!":"Unsere Rakete wächst."}</h2><p>{komplett?"Vier Lernorte. Zwölf Ideen. Zeit für den gemeinsamen Abschluss.":"In jedem Lernort warten drei Aufgaben. Ein fertiges Kapitel schenkt uns ein Raketenteil."}</p></div>{komplett&&<button className="welt-knopf" onClick={()=>setAnsicht("finale")}>Zum Raketenstart</button>}</section><p className="reise-kleindruck">Echte Stadtpositionen, gemalte Ansichten. Gebäude und Umgebung sind vereinfacht dargestellt.</p></section>}
 {ansicht==="ort"&&<section className="reise-ort"><button className="welt-chip zur-stadt" onClick={()=>setAnsicht("stadt")}>← Zur Stadt</button><div className="ort-buehne"><img className="ort-kulisse" src={ort.bild} alt={`Gemalte Außenansicht: ${ort.name}, ${ort.adresse}, Heilbronn`}/><div className="ort-schild"><p>HEILBRONN · {ort.adresse}</p><h1>{ort.name}</h1></div><div className="ort-guide"><Avatar/><div className="guide-sprechblase"><p>{ort.beschreibung}</p><Sprachknopf id={`ort-${ort.id}`} stumm={stumm}/></div></div></div><div className="ort-aufgaben"><h2>Was probieren wir aus?</h2><div>{[0,1,2].map(n=><button key={n} className="ort-aufgabe" onClick={()=>{neuePhase(n);setAnsicht("aufgabe");}}><span>{erfolge.includes(`${ort.id}-${n}`)?"✓":n+1}</span><strong>{AUFGABEN_TITEL[ort.id]![n]}</strong><b aria-hidden="true">→</b></button>)}</div></div><p className="reise-kleindruck">Eigene Lernspiele von COSMOS. Die Illustration ist eine vereinfachte Architekturansicht, kein Gebäudeplan.</p></section>}
 {ansicht==="aufgabe"&&<section className="reise-aufgabe"><header><button className="welt-chip" onClick={()=>setAnsicht("ort")}>← {ort.name}</button><span>Idee {phase+1} / 3 · {STUFEN.find(s=>s.id===stufe)?.name}</span><button className="welt-chip" onClick={()=>setEingefuehrt(false)}>Erklärung</button></header><div className="aufgaben-raum" style={{"--raum-bild":`url(${ort.bild})`} as React.CSSProperties}><div className="aufgaben-papier">{!eingefuehrt?<AufgabenEinstieg art={station.aufgabe} stufe={stufe} phase={phase} stumm={stumm} starten={()=>{setEingefuehrt(true);setErfolg(false);}}/>:<><Lernaufgabe key={`${ort.id}-${stufe}-${phase}`} art={station.aufgabe} stufe={stufe} phase={phase} geschafft={geschafft}/>{erfolg&&<div className="reise-erfolg" role="status"><span className="erfolgs-blume" aria-hidden="true">✓</span><h2>Eine neue Idee für unsere Rakete!</h2><div className="welt-aktionen"><button className="welt-knopf" onClick={()=>phase<2?neuePhase(phase+1):setAnsicht("stadt")}>{phase<2?"Nächste Idee":"Zurück nach Heilbronn"}</button><button className="welt-knopf hell" onClick={()=>setAnsicht("stadt")}>Zur Stadtübersicht</button></div></div>}</>}</div></div></section>}
 {ansicht==="finale"&&<section className="reise-finale"><p className="welt-augenbraue">HEILBRONN STECKT VOLLER IDEEN</p><h1>Bereit für neue Entdeckungen.</h1><div className="finale-illustration"><Avatar/><svg viewBox="-80 -130 160 190" aria-label="Unsere vollständige Rakete"><Rakete teile={teile} zustand="bereit"/></svg></div><p>Du hast programmiert, geforscht, gebaut und genau hingeschaut.</p><h2>Was möchtest du in echt ausprobieren?</h2><div className="welt-aktionen">{WUENSCHE.map(w=><button className="welt-chip" key={w.id} aria-pressed={stand.wuensche.includes(w.id)} onClick={()=>wunsch(w.id)}>{w.titel}</button>)}</div><div className="welt-aktionen"><button className="welt-knopf" onClick={erwachsene}>Meiner Familie zeigen</button><button className="welt-knopf hell" onClick={()=>setAnsicht("stadt")}>Weiter entdecken</button></div></section>}
 </>}
 <footer className="reise-fuss"><span>Ohne Namen. Ohne Konto. Gemeinsam entdecken.</span><div><button onClick={()=>setStumm(!stumm)} aria-pressed={stumm}>Ton {stumm?"aus":"an"}</button><button onClick={bildschirm}>{vollbild?"Verkleinern":"Vollbild"}</button><button onClick={erwachsene}>Eltern & Datenschutz</button></div></footer>
 <dialog ref={dialog} className="reise-dialog" onCancel={()=>{haltenEnde();setFenster(undefined);}} aria-labelledby="reise-dialog-titel"><header><h2 id="reise-dialog-titel">{fenster==="tor"?"Für Erwachsene":fenster==="erwachsene"?"Gemeinsam weiterentdecken":fenster==="hilfe"?"So geht’s":"Deine Kursideen"}</h2><button className="welt-chip" onClick={()=>{haltenEnde();setFenster(undefined);}} aria-label="Fenster schließen">×</button></header>
 {fenster==="tor"&&<div className="erwachsenen-tor"><p>Hier geht es zu Kursen, Einstellungen und externen Seiten.</p><p>Halte den Knopf drei Sekunden gedrückt. Mit der Tastatur: Leertaste oder Enter halten.</p><button className={`welt-knopf halteknopf ${halten?"haelt":""}`} onPointerDown={e=>{e.currentTarget.setPointerCapture(e.pointerId);haltenStart();}} onPointerUp={haltenEnde} onPointerCancel={haltenEnde} onLostPointerCapture={haltenEnde} onKeyDown={e=>{if((e.key===" "||e.key==="Enter")&&!e.repeat){e.preventDefault();haltenStart();}}} onKeyUp={e=>{if(e.key===" "||e.key==="Enter"){e.preventDefault();haltenEnde();}}} onBlur={haltenEnde}>{halten?"Bitte halten …":"Für Erwachsene öffnen"}</button><small>Schutz vor versehentlichem Öffnen; keine Altersprüfung.</small></div>}
 {fenster==="erwachsene"&&<section className="erwachsenen-inhalt"><div className="welt-aktionen"><Link href="/eltern/#spielbericht" className="welt-knopf">Entdeckungen & Kurse</Link><Link href="/einrichtungen/" className="welt-knopf hell">Originalorte & Socials</Link><Link href="/datenschutz/" className="welt-knopf hell">Datenschutz & Löschen</Link><Link href="/" className="welt-knopf hell">Zur Startseite</Link></div><fieldset className="reise-stufen"><legend>Aufgabenstufe</legend>{STUFEN.map(s=><button key={s.id} className="welt-chip" aria-pressed={stufe===s.id} onClick={()=>{{setKursalterBestaetigt(false);speichere({...stand,alter:s.min===6?8:s.min===9?10:14});};setErfolg(false);setEingefuehrt(false);}}>{s.name}<small>{s.alter}</small></button>)}</fieldset><label className="kursalter">Genaues Alter nur für passende Kurse<select aria-label="Alter für Kursvorschläge" value={kursalterBestaetigt?stand.alter:""} onChange={e=>{setKursalterBestaetigt(true);speichere({...stand,alter:Number(e.target.value)});}}><option value="" disabled>Alter wählen</option>{Array.from({length:12},(_,n)=><option key={n} value={n+6}>{n+6} Jahre</option>)}</select></label><details><summary>Zu den Illustrationen und Sprachaufnahmen</summary><p>Die Karte verwendet lokale OpenStreetMap-Daten. Die vier Szenen sind vorab generierte, vereinfachte Architekturillustrationen. Kein exaktes Gebäudemodell, keine bestätigten Innenraumansichten. Die Aufgabenansichten sind frei gestaltete Lernflächen.</p><p>Die festen Sprachaufnahmen sind noch nicht produziert. Der Player ist vorbereitet; es gibt keinen versteckten Ersatz durch Browserstimmen. Alle Bildanleitungen funktionieren ohne Ton.</p><a href={ort.quelle} target="_blank" rel="noopener noreferrer">Offizielle Quelle zum ausgewählten Ort</a></details><button className="welt-knopf hell" onClick={()=>setReset(!reset)}>Spiel zurücksetzen</button>{reset&&<div className="reset-frage"><p>Figur, alle Aufgabenstufen, Wünsche und Kinder-Rucksack in dieser Sitzung löschen?</p><button className="welt-knopf" onClick={neu}>Ja, neu beginnen</button><button className="welt-knopf hell" onClick={()=>setReset(false)}>Behalten</button></div>}<h3>Echte Kurse zum ausgewählten Thema</h3><p>{station.id==="experimenta"?"Die folgenden Coding-Angebote sind zusätzliche Themenideen. Das Stromrätsel ist kein Eignungstest.":"Thema und Kursalter bestimmen diese Auswahl. Teilnahmebedingungen beim Veranstalter prüfen."}</p>{passende.length?<div className="spiel-kurskarten">{passende.slice(0,6).map(karte)}</div>:<p>{kursalterBestaetigt?"Im gespeicherten Bestand gibt es dafür gerade keinen passenden Kurs.":"Wähle zuerst das konkrete Kursalter. Eine Spielstufe ersetzt keine Altersangabe."}</p>}</section>}
 {fenster==="hilfe"&&<div className="reise-hilfe"><Avatar/><ol><li>Gestalte deine eigene Figur.</li><li>Tippe einen Ort auf der Heilbronn-Karte an.</li><li>Schau die Erklärung an. Probiere die Aufgabe aus.</li><li>Drei Aufgaben ergeben ein Raketenteil.</li></ol><button className="welt-knopf" onClick={()=>setFenster(undefined)}>Verstanden</button></div>}
 {fenster==="rucksack"&&<div className="erwachsenen-inhalt"><p>{merk.ids.length?"Deine gemerkten Angebote schaust du zusammen mit einem Erwachsenen an.":"Dein Rucksack ist noch leer. Echte Kurse kannst du gemeinsam mit einem Erwachsenen auswählen."}</p><button className="welt-knopf" onClick={erwachsene}>Gemeinsam anschauen</button></div>}
 </dialog></main></AvatarKontext.Provider>;
}
